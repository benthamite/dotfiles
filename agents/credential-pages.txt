# Pages that display credentials. Screenshots taken while a browser tab is on
# one of these are stripped from tool output before the model sees them
# (claude/hooks/redact-tool-output.py, codex/hooks/redact-tool-output.py).
# Five of the nine browser exposures in docs/secret-guard.md were screenshots.
#
# One Python regular expression per line, matched with re.search against the
# full URL (scheme, host, path, query). Add a provider's key, token or OAuth
# settings page here when an agent may visit it. Blank lines and # comments
# are ignored.

# 1Password web app: every page can show vault contents.
^https?://[^/]*\.1password\.(com|ca|eu)/
# Anthropic / Claude console
^https?://(console\.anthropic\.com|platform\.claude\.com)/settings/(keys|admin-keys)
# OpenAI
^https?://platform\.openai\.com/(api-keys|settings/.*(api-keys|admin-keys|service-accounts))
# Google
^https?://console\.cloud\.google\.com/(apis/credentials|auth/clients|iam-admin/serviceaccounts)
^https?://aistudio\.google\.com/(app/)?apikey
# GitHub
^https?://github\.com/settings/(tokens|personal-access-tokens|apps|developers|keys|security)
^https?://github\.com/[^/]+/[^/]+/settings/(secrets|keys|hooks)
# Slack apps: credentials, OAuth tokens, install pages
^https?://api\.slack\.com/apps/[^/]+/(general|oauth|install-on-team|event-subscriptions)
# Cloudflare
^https?://dash\.cloudflare\.com/(profile/api-tokens|[^/]+/api-tokens|.*/r2/api-tokens)
# Netlify, Vercel, AWS
^https?://app\.netlify\.com/user/applications
^https?://vercel\.com/account/tokens
^https?://[^/]*console\.aws\.amazon\.com/iam/.*(security_credentials|users/.*security)
